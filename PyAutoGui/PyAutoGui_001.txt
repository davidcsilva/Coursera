#Tarefa com horario marcado
import pyautogui
import time
import datetime 

#hora de inicio
hora_de_inicio = datetime.time(20,30)

#Pega o horario atual
agora = datetime.datetime.now()

#cria um datetime para o horario de inicio

alvo = datetime.datetime.combine(agora.date(),hora_de_inicio)

#Calcula o tempo de espera

segundos_para_esperar = (alvo - agora).total_seconds()

# Execução da tarefa
time.sleep(segundos_para_esperar)
